cmake_minimum_required(VERSION 3.18)
project(s2e_bridges LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
foreach(v NRN_SOURCE_DIR NRN_BUILD_DIR NRN_INSTALL_DIR)
  if(NOT DEFINED ${v})
    message(FATAL_ERROR "Supply -D${v}=absolute/path")
  endif()
endforeach()
find_package(Python COMPONENTS Interpreter Development REQUIRED)
find_package(pybind11 CONFIG REQUIRED)
set(INCLUDES ${NRN_BUILD_DIR} ${NRN_SOURCE_DIR}/src ${NRN_SOURCE_DIR}/external/Random123/include)
foreach(d oc nrnoc ivoc nrncvode nrniv modlunit nmodl ivos neuronmusic gnu mesch nrnmpi nrnpython parallel scopmath sparse sparse13 sundials sundials/cvodes sundials/ida sundials/shared)
 list(APPEND INCLUDES ${NRN_SOURCE_DIR}/src/${d} ${NRN_BUILD_DIR}/src/${d})
endforeach()
foreach(target data_c2py data_c2py_v2)
 pybind11_add_module(${target} ../src/bridges/${target}.cpp)
 target_include_directories(${target} PRIVATE ${INCLUDES})
 target_link_libraries(${target} PRIVATE ${NRN_INSTALL_DIR}/lib/libnrniv.so)
 set_target_properties(${target} PROPERTIES BUILD_RPATH ${NRN_INSTALL_DIR}/lib)
endforeach()
